  %% сигнал: источники в пре-процессинг
  LOW --> DRONE_FLTR
  MID --> DRONE_FLTR
  HIGH --> DRONE_FLTR
  LEAD --> LEAD_TONE
  MIC --> MIC_IN

  %% пре-процессинг: панорама и уровни
  DRONE_FLTR --> PAN_CAPTURE
  LEAD_TONE --> PAN_CAPTURE

  %% микширование: сухая сумма и шина отправок
  PAN_CAPTURE --> SUM
  PAN_CAPTURE -->|"pre-fader"| SEND_BUS
  PERC -->|"post-fader"| SUM
  PERC -->|"post-fader"| SEND_BUS
  MIC_IN -->|"AUX level 0"| SEND_BUS

  %% гейт входа, затем эффекты и посылка
  SUM --> GATE
  SEND_BUS --> GATE
  GATE --> CHORUS
  GATE -->|"посылка"| VERB_SEND
  CHORUS --> DELAY
  CHORUS -->|"wet"| VERB_SEND
  DELAY -->|"wet"| VERB_SEND

  %% реверберация: посылка/возврат и баланс
  VERB_SEND -->|"ALGO: ROOM · HALL · CATHEDRAL · PLATE · HOLD"| VERB
  VERB -->|"wet"| VERB_MIX
  DELAY -->|"сухой"| VERB_MIX

  %% мастер-цепочка
  VERB_MIX --> COMP
  COMP --> VOL
  VOL --> SOFT_CLIP
  SOFT_CLIP -->|"INT16 · I2S"| PCM5100A
